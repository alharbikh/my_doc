4) أمثلة عملية للكود

import { Document, Page, Text, View, StyleSheet } from '@react-pdf/renderer'; const styles = StyleSheet.create({ page: { padding: 30 }, header: { fontSize: 18, bold: true, marginBottom: 20 }, row: { flexDirection: 'row', marginBottom: 10 }, label: { width: 100, fontWeight: 'bold' }, value: { flex: 1 }, }); const SchoolCommitteePDF = ({ committeeData }) => ( <Document> <Page style={styles.page}> <Text style={styles.header}> قرار تشكيل لجنة: {committeeData.name} </Text> {committeeData.members.map((member, index) => ( <View key={index} style={styles.row}> <Text style={styles.label}>العضو {index + 1}:</Text> <Text style={styles.value}>{member.name}</Text> </View> ))} </Page> </Document> ); // Usage <PDFDownloadLink document={<SchoolCommitteePDF committeeData={data} />} fileName="committee.pdf" > تحميل القرار </PDFDownloadLink>

📜 jsPDF Example

نفس النموذج باستخدام jsPDF:

import jsPDF from 'jspdf'; const generateCommitteePDF = (committeeData) => { const doc = new jsPDF(); let y = 20; // Header doc.setFontSize(18); doc.setFont('helvetica', 'bold'); doc.text(`قرار تشكيل لجنة: ${committeeData.name}`, 20, y); y += 30; // Members doc.setFontSize(12); doc.setFont('helvetica', 'normal'); committeeData.members.forEach((member, index) => { doc.text(`العضو ${index + 1}:`, 20, y); doc.text(member.name, 100, y); y += 15; }); // Save doc.save('committee.pdf'); }; // Usage generateCommitteePDF(data);
